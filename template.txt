<script>
  // â”€â”€ DATA â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  const SESSIONS = %INJECT_DATA%;

  // â”€â”€ STATE â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  let watched    = JSON.parse(localStorage.getItem('tc-ct-watched') || '{}');
  let autoNext   = false;
  let activeItem = null;
  let currentVideoId = null;
  let inlinePlayer   = null;
  let ytPlayer, ytReady = false;
  const playQueue = [];

  // â”€â”€ YOUTUBE API â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  function onYouTubeIframeAPIReady() {
    ytPlayer = new YT.Player('yt-player', {
      height:'100%', width:'100%',
      playerVars:{ rel:0, modestbranding:1, autoplay:1 },
      events:{
        onReady() { ytReady = true; },
        onStateChange(e) {
          if (e.data === YT.PlayerState.ENDED) {
            if (activeItem) markWatched(currentVideoId, activeItem);
            if (autoNext) playNext();
          }
          if (e.data === YT.PlayerState.PLAYING && activeItem) {
            markWatched(currentVideoId, activeItem);
          }
        }
      }
    });
  }

  function embedUrl(id) {
    return 'https://www.youtube-nocookie.com/embed/' + id + '?rel=0&modestbranding=1&autoplay=1';
  }

  // â”€â”€ PLAY â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  function playVideo(id, name, sessionLabel, el) {
    closeInline(false);
    if (activeItem) activeItem.classList.remove('playing');
    el.classList.add('playing');
    activeItem = el; currentVideoId = id;

    if (window.innerWidth >= 1024) { // lg breakpoint in tailwind
      document.getElementById('vid-placeholder').style.display = 'none';
      const ypEl = document.getElementById('yt-player');
      ypEl.style.display = 'block';
      if (ytReady && ytPlayer && ytPlayer.loadVideoById) {
        ytPlayer.loadVideoById(id);
      } else {
        ypEl.innerHTML = `<iframe src="${embedUrl(id)}" allow="accelerometer;autoplay;clipboard-write;encrypted-media;gyroscope;picture-in-picture" allowfullscreen style="position:absolute;top:0;left:0;width:100%;height:100%;border:none"></iframe>`;
      }
      document.getElementById('np-session').textContent = sessionLabel;
      document.getElementById('np-title').textContent = name;
      document.getElementById('np-desc').textContent = "ChÃºc bạn học táº­p hiệu quả!";
    } else {
      markWatched(id, el);
      const div = document.createElement('div');
      div.className = 'inline-player';
      div.innerHTML = `<iframe src="${embedUrl(id)}" allow="accelerometer;autoplay;clipboard-write;encrypted-media;gyroscope;picture-in-picture" allowfullscreen frameborder="0"></iframe>`;
      el.after(div);
      inlinePlayer = div;
      setTimeout(() => div.scrollIntoView({ behavior:'smooth', block:'center' }), 100);
    }
  }

  function playNext() {
    const idx = playQueue.findIndex(v => v.id === currentVideoId);
    if (idx >= 0 && idx < playQueue.length - 1) {
      const next = playQueue[idx + 1];
      const sessEl = next.el.closest('.session');
      if (sessEl && sessEl.classList.contains('closed')) toggleSession(sessEl.querySelector('.session-title'));
      next.el.scrollIntoView({ behavior:'smooth', block:'nearest' });
      playVideo(next.id, next.name, next.session, next.el);
    }
  }

  function closeInline(clearActive = true) {
    if (inlinePlayer) { inlinePlayer.remove(); inlinePlayer = null; }
    if (clearActive && activeItem) {
      activeItem.classList.remove('playing');
      activeItem = null;
    }
  }

  // â”€â”€ PROGRESS â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  function markWatched(id, el) {
    if (!id || watched[id]) return;
    watched[id] = true;
    localStorage.setItem('tc-ct-watched', JSON.stringify(watched));
    if (el) {
      el.classList.add('watched');
    }
    updateStats();
  }

  function applyWatchedState() {
    Object.keys(watched).forEach(id => {
      document.querySelectorAll(`[data-vid="${id}"]`).forEach(el => {
        el.classList.add('watched');
      });
    });
  }

  function updateStats() {
    const total = playQueue.length;
    const done  = Object.keys(watched).length;
    const pct   = total > 0 ? Math.round(done / total * 100) : 0;
    const el    = document.getElementById('ts-watched-text');
    if (el) el.textContent = `HoÃ n thÃ nh: ${pct}%`;
  }

  function toggleAutoNext() {
    autoNext = !autoNext;
    const btn = document.getElementById('btn-autonext');
    if (autoNext) {
        btn.style.background = 'linear-gradient(135deg, #a8ff78, #00f2fe)';
        btn.innerHTML = 'Tá»± Ä‘á»™ng: Báº¬T <span class="text-lg">âœ“</span>';
    } else {
        btn.style.background = 'linear-gradient(135deg, #00f2fe, #4facfe)';
        btn.innerHTML = 'Tá»± Ä‘á»™ng: Táº®T <span class="text-lg">â–¶</span>';
    }
  }

  // â”€â”€ BUILD SESSIONS â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  function toggleSession(titleEl) {
    const sessionEl = titleEl.closest('.session');
    sessionEl.classList.toggle('closed');
  }

  function buildSessions() {
    const container = document.getElementById('playlist-container');
    let totalV = 0;
    let html = '';

    SESSIONS.forEach((s, si) => {
      const hasVideos = s.videos.length > 0;
